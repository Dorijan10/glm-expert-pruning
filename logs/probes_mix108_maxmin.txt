```python
import numpy as np

def pairwise_euclidean_distances(X, Y):
    """
    Compute pairwise Euclidean distances between two sets of points X and Y.
    
    Parameters:
    X : ndarray with shape (N, D)
    Y : ndarray with shape (M, D)
    
    Returns:
    D : ndarray with shape (N, M)
        where D[i, j] is the Euclidean distance between X[i] and Y[j]
    """
    # Compute squared distances using broadcasting
    # X[:, np.newaxis, :] has shape (N, 1, D)
    # Y[np.newaxis, :, :] has shape (1, M, D)
    # Their difference has shape (N, M, D)
    diff = X[:, np.newaxis, :] - Y[np.newaxis, :, :]
    
    # Square the differences, sum along the last axis (D), and take the square root
    D = np.sqrt(np.sum(diff ** 2, axis=-1))
    
    return D
```

```python
    """
    Merges two sorted vectors into a single sorted vector.

        vec1: vec2: sorted lists (or vectors)
```
